# Co-change claims on the corpus (2026-10-11)

What git history says ships together, measured to choose the co-change claim's settings and then
re-run with the build's own walk and gates. DECISIONS C55, D14 and F27 cite it.

## Method

- Repositories: the 35 in `anatomiya-corpus`, and the nine Go, PHP and Python ones in
  `anatomiya-corpus-langs` (caddy, gin, hugo, composer, laravel, Slim, django, fastapi, flask). Each
  was copied with `cp -Rc`, measured, and deleted. The nine are shallow clones of 400 to 579
  commits; they were measured as cloned, and again on copies deepened with `git fetch
  --deepen=20000`.
- Walk: `git -c log.showRoot=false log <tip> --first-parent --diff-merges=first-parent --no-renames
  --name-status -z --format=%x1e%H%x1f%aE -n <depth> --`, where the tip is HEAD's fork point with
  the first of `origin/HEAD`, `origin/main`, `origin/master`, `main`, `master` that resolves. No
  copy held a pin.
- A change is one first-parent commit. Changes with no added or modified file are empty; changes
  with more than 40 are bulk; both are left out. A path added under a basename the same change
  deletes is an edit.
- A class is a directory plus a suffix (from the first dot, past segments starting with a digit;
  `_spec.rb` and `_test.rb` whole), and the same one directory up.
- Mined on the older four fifths of the changes read, held out on the newest fifth. "Mined over all"
  is what a scan would state.
- Gates: test class on neither side; the companion one named file or a flat class; ratio 0.90;
  Wilson lower bound 0.90; 2 authors over conforming candidates; lift at least 2; the companion and
  a producer exist at the tip; the newest fifth of the candidates (at least 5) at 0.90. One
  statement per pair.
- Machine load during the runs: 11 to 44 on the one-minute average (other jobs). Times are wall
  clock for the git calls and the counting, and include that load.

```js
// Probe for the co-change design: mines "adding a file of class A ships with touching B" from
// first-parent history the way the design specifies, and tests the rules on the newest fifth.
// usage: node cochange-probe.mjs <repo> [--units N] [--json out.json]
import { execFileSync } from "node:child_process";
import { writeFileSync } from "node:fs";
import { basename, dirname } from "node:path";
import { wilsonLower } from "<anatomiya>/plugins/anatomiya/lib/gates.mjs";

const args = process.argv.slice(2);
const repo = args[0];
const opt = (name, fallback) => {
  const i = args.indexOf(name);
  return i === -1 ? fallback : args[i + 1];
};
const UNITS = Number(opt("--units", 4000));
const OUT = opt("--json", null);
const UNIT_MOST = 40;
const RATIO = 0.9;
const LIFT = 2;
const AUTHORS = 2;
const PROBE_SUPPORT = 20;
const CAP = Number(opt("--cap", 100)); // the probe's own gate, kept beside the product's for comparison

const git = (...a) => execFileSync("git", ["-C", repo, ...a], { encoding: "utf8", maxBuffer: 1 << 30, stdio: ["ignore", "pipe", "ignore"] });
const tryGit = (...a) => {
  try {
    return git(...a).trim();
  } catch {
    return null;
  }
};

// Tip: the pin where there is one, else HEAD's fork point with the default branch, else HEAD.
const BASE_REFS = ["origin/HEAD", "origin/main", "origin/master", "main", "master"];
let tip = null;
let tipFrom = null;
const head = tryGit("rev-parse", "--verify", "-q", "HEAD^{commit}");
for (const ref of BASE_REFS) {
  const sha = tryGit("rev-parse", "--verify", "-q", `${ref}^{commit}`);
  if (!sha) continue;
  tip = tryGit("merge-base", head, sha) ?? sha;
  tipFrom = `fork point with ${ref}`;
  break;
}
if (!tip) {
  tip = head;
  tipFrom = "HEAD";
}
const shallow = tryGit("rev-parse", "--is-shallow-repository") === "true";

const t0 = performance.now();
const raw = git("-c", "log.showRoot=false", "log", tip, "--first-parent", "--diff-merges=first-parent", "--no-renames", "--name-status", "-z", "--format=%x1e%H%x1f%aE", "-n", String(UNITS), "--");
const walkMs = performance.now() - t0;

const t1 = performance.now();
const atTip = new Set(git("ls-tree", "-r", "-z", "--name-only", tip).split("\0").filter(Boolean));
const treeMs = performance.now() - t1;

function suffixOf(base) {
  const ruby = base.match(/(_spec|_test)\.rb$/);
  if (ruby) return ruby[0];
  const parts = base.split(".");
  let k = 1;
  if (parts[0] === "") k = 2; // a dotfile's own name is not its suffix
  if (parts.length <= k) return null;
  // A version number is not a kind of file: `foo.1.0.gem` is a `.gem`.
  while (k < parts.length - 1 && /^\d/.test(parts[k])) k++;
  return `.${parts.slice(k).join(".")}`;
}

function classesOf(path) {
  const dir = dirname(path);
  const suffix = suffixOf(basename(path));
  if (suffix === null) return [];
  const out = [`${dir === "." ? "" : `${dir}/`}*${suffix}`];
  if (dir.includes("/")) out.push(`${dirname(dir)}/*/*${suffix}`);
  return out;
}

const t2 = performance.now();
const units = [];
let bulk = 0;
let empty = 0;
for (const record of raw.split("\x1e")) {
  if (!record.trim()) continue;
  const fields = record.split("\0");
  const [, email = ""] = fields[0].split("\x1f");
  const author = email.trim().toLowerCase();
  const files = [];
  const gone = new Set();
  for (let i = 1; i + 1 < fields.length; i += 2) {
    const status = fields[i].replace(/^\n+/, "")[0];
    const path = fields[i + 1];
    if (!status || !path) continue;
    if (status === "A" || status === "M" || status === "T") files.push({ status: status === "T" ? "M" : status, path });
    else if (status === "D") gone.add(basename(path));
  }
  for (const f of files) if (f.status === "A" && gone.has(basename(f.path))) f.status = "M";
  if (files.length === 0) {
    empty++;
    continue;
  }
  if (files.length > UNIT_MOST) {
    bulk++;
    continue;
  }
  const triggers = new Set();
  const touched = new Set();
  const added = [];
  for (const f of files) {
    touched.add(`edits ${f.path}`);
    if (f.status === "A") {
      added.push(f.path);
      for (const c of classesOf(f.path)) {
        triggers.add(c);
        touched.add(`adds ${c}`);
      }
    }
  }
  units.push({ author, triggers, touched, added });
}

const isTestClass = (c) =>
  /(^|\/|\s)(spec|specs|test|tests|__tests__|__snapshots__|e2e|fixtures?|__fixtures__)\//i.test(c) ||
  /(_spec|_test|\.spec|\.test|\.stories|Test|Tests|\.snap)(\.[\w]+)*$/.test(c) ||
  /\/test_[^/]*$/.test(c);
const isPerson = (e) => !e.includes("[bot]");
const classMatch = (cls, path) => classesOf(path).includes(cls);
const filesOfClass = (cls) => {
  let n = 0;
  for (const p of atTip) if (classMatch(cls, p)) n++;
  return n;
};

function mine(train, { product, wilson = product, cap = Infinity }) {
  const support = new Map();
  const together = new Map();
  const base = new Map();
  // Each producer class is counted over its newest `cap` candidates only.
  const seenT = new Map();
  const live = train.map((u) => {
    const keep = new Set();
    for (const t of u.triggers) {
      const k = seenT.get(t) ?? 0;
      if (k < cap) keep.add(t);
      seenT.set(t, k + 1);
    }
    return keep;
  });
  for (const [ui, u] of train.entries()) {
    for (const c of u.touched) base.set(c, (base.get(c) ?? 0) + 1);
    for (const t of live[ui]) {
      support.set(t, (support.get(t) ?? 0) + 1);
      for (const c of u.touched) {
        if (c === `adds ${t}`) continue;
        together.set(`${t}\x01${c}`, (together.get(`${t}\x01${c}`) ?? 0) + 1);
      }
    }
  }
  const rules = [];
  const gated = { ratio: 0, evidence: 0, authors: 0, lift: 0, exists: 0, recency: 0, test: 0, shape: 0 };
  for (const [key, n] of together) {
    const [t, c] = key.split("\x01");
    const of = support.get(t);
    if (n / of < RATIO || of < (product && wilson ? 10 : PROBE_SUPPORT)) continue; // near-miss floor
    const r = { t, c, n, of };
    // Shape: the companion is one named file, or a flat class.
    if (c.startsWith("edits ") ? false : c.includes("/*/*")) { gated.shape++; continue; }
    const bareC = c.replace(/^(adds|edits) /, "");
    if (related(t, bareC)) { gated.shape++; continue; }
    if (product) {
      // Tests are the namesake count's and the obligations' to speak for, file by file.
      if (isTestClass(t) || isTestClass(bareC)) { gated.test++; continue; }
      if (wilson && wilsonLower(n, of) < RATIO) { gated.evidence++; continue; }
    }
    // Authors over the conforming candidates.
    const hands = new Set();
    const cands = [];
    for (const [ui, u] of train.entries()) {
      if (!live[ui].has(t)) continue;
      const ok = u.touched.has(c);
      cands.push(ok);
      if (ok && isPerson(u.author)) hands.add(u.author);
    }
    r.authors = hands.size;
    if (hands.size < AUTHORS) { gated.authors++; continue; }
    const lift = n / of / (base.get(c) / train.length);
    r.lift = lift;
    if (lift < LIFT) { gated.lift++; continue; }
    if (product) {
      // Existence at the tip: the named file, or one file of the companion class, and one producer.
      const exists = c.startsWith("edits ") ? atTip.has(bareC) : [...atTip].some((p) => classMatch(bareC, p));
      const producer = [...atTip].some((p) => classMatch(t, p));
      if (!exists || !producer) { gated.exists++; continue; }
      // Recency: the newest fifth of the candidates, at least five, still reaches the ratio.
      const k = Math.max(5, Math.ceil(cands.length / 5));
      const newest = cands.slice(0, k);
      const met = newest.filter(Boolean).length;
      r.recent = `${met} of ${newest.length}`;
      if (met / newest.length < RATIO) { gated.recency++; continue; }
    }
    rules.push(r);
  }
  return { rules: onePerPair(rules), gated, pairKeys: together.size, prunedKeys: pruned(train) };
}

// How many pair keys a two-pass count holds: triggers and companions each seen in enough units first.
function pruned(train) {
  const seen = new Map();
  const trig = new Map();
  for (const u of train) {
    for (const c of u.touched) seen.set(c, (seen.get(c) ?? 0) + 1);
    for (const t of u.triggers) trig.set(t, (trig.get(t) ?? 0) + 1);
  }
  const keys = new Set();
  for (const u of train) {
    const ts = [...u.triggers].filter((t) => trig.get(t) >= 35);
    if (!ts.length) continue;
    const cs = [...u.touched].filter((c) => seen.get(c) >= 32);
    for (const t of ts) for (const c of cs) keys.add(`${t}\x01${c}`);
  }
  return keys.size;
}

const up = (cls) => {
  const m = cls.match(/^(.*)\/\*(\.[^/]*|_spec\.rb|_test\.rb)$/);
  if (!m || m[1].endsWith("/*") || !m[1].includes("/")) return null;
  return `${dirname(m[1])}/*/*${m[2]}`;
};
const related = (a, b) => a === b || up(a) === b || up(b) === a;

function onePerPair(rules) {
  // The directory's own class over the wildcard above it.
  let keep = rules.filter((r) => !(r.t.includes("/*/*") && rules.some((o) => o !== r && o.c === r.c && up(o.t) === r.t && o.of >= r.of * 0.9)));
  // Both directions: keep the one whose producer is not a test, else the larger, else by name.
  keep = keep.filter((r) => {
    if (!r.c.startsWith("adds ")) return true;
    const back = keep.find((o) => o.t === r.c.slice(5) && o.c === `adds ${r.t}`);
    if (!back) return true;
    const rt = isTestClass(r.t);
    const bt = isTestClass(back.t);
    if (rt !== bt) return !rt;
    if (r.of !== back.of) return r.of > back.of;
    return r.t < back.t;
  });
  return keep;
}

function holdout(rules, held) {
  let fired = 0;
  let met = 0;
  for (const r of rules) {
    r.fired = 0;
    r.met = 0;
    for (const u of held) {
      if (!u.triggers.has(r.t)) continue;
      r.fired++;
      if (u.touched.has(r.c)) r.met++;
    }
    fired += r.fired;
    met += r.met;
  }
  return { fired, met };
}
```
